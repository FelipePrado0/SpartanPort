cmake_minimum_required(VERSION 3.20)
project(spartanport_streamline LANGUAGES CXX)
set(STREAMLINE_SDK_ROOT "" CACHE PATH "Pinned Streamline 2.14.16 SDK")
set(VULKAN_INCLUDE "" CACHE PATH "Vulkan-Headers include directory")
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Streamline bridge requires MSVC x64")
endif()
if(NOT EXISTS "${STREAMLINE_SDK_ROOT}/include/sl_dlss_g.h")
    message(FATAL_ERROR "Set STREAMLINE_SDK_ROOT to the pinned Streamline SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_library(spartanport_streamline SHARED spartanport_streamline.cpp)
target_include_directories(spartanport_streamline PRIVATE "${STREAMLINE_SDK_ROOT}/include" "${VULKAN_INCLUDE}")
target_compile_definitions(spartanport_streamline PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(spartanport_streamline PRIVATE bcrypt)
add_executable(streamline-bridge-test EXCLUDE_FROM_ALL ../../tests/test_streamline_bridge.cpp)
target_include_directories(streamline-bridge-test PRIVATE "${STREAMLINE_SDK_ROOT}/include" "${VULKAN_INCLUDE}")
target_compile_definitions(streamline-bridge-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(streamline-bridge-test PRIVATE bcrypt)
